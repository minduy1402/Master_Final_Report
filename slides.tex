\begin{frame}{Other Technical Barriers}
    \begin{itemize}
        \item \textbf{Strict data security constraints:} Sensitive and private business data can not be exposed to ChatGPT or Cloud Services
        \item \textbf{High accuracy requirements:} Hallucination is unacceptable in high-stakes domains like finance and healthcare
        \item \textbf{Resource-constrained deployment:} Large models require significant computational resources, while smaller models have limited context windows
    \end{itemize}
    
    \vspace{0.5cm}
    \textbf{\color{MainNavy} $\rightarrow$ Solution Objective:} A unified in-house RAG pipeline that can handle complex documents with high accuracy
\end{frame}

\begin{frame}{Online Querying Phase}
    \begin{figure}
        \centering
        \includegraphics[width=0.65\textwidth]{figures/querying2.png}
    \end{figure}

    \footnotesize
    \textbf{Combining Two Approaches:}
    \begin{itemize}
        \item \textbf{BM25 (Keyword):} Exact term matching for codes, names
        \item \textbf{Vector Search (Semantic):} Contextual understanding
        \item \textbf{Reciprocal Rank Fusion:} Merge results intelligently
    \end{itemize}

    \textbf{Reranking Process:}
    \begin{enumerate}
        \item Hybrid search retrieves candidate documents
        \item Cross-encoder computes relevance scores for each query-document pair
        \item Top-k most relevant documents sent to LLM for generation
    \end{enumerate}
    
    \textbf{Benefit:} Addresses "lost in the middle" problem by ensuring most relevant context is highly ranked
\end{frame}

\begin{frame}{Problem 2: Behavior Adaptation for Small Models}
    \textbf{Context:} Building a chatbot for Tarot Reading (psychological counseling)
    
    \vspace{0.3cm}
    \textbf{Naive Approach:} Prompt engineering with large LLMs (GPT-4, Claude-3) via APIs
    
    \vspace{0.3cm}
    \textbf{Why Prompt Engineering Alone Is Insufficient:}
    \begin{itemize}
